# -*- coding: utf-8 -*-
# שלב 1 — חילוץ פורנזי מלא (PyMuPDF בלבד, ללא VLM):
# לכל PDF: פליטת spans (כולל כיוון כתיבה dir), ציורים וקטוריים, תמונות, והיסטוגרמת גופן@גודל
import fitz, json, os
OUT = "/home/z/my-project/scripts/out/pdf3"
FILES = {"old": ".../PDF File Export - Old Engine.pdf",
         "new": ".../PDF File Export - New Engine - 23 11 2026 08 30.pdf"}

def dump(tag):
    doc = fitz.open(FILES[tag])
    with open(f"{OUT}/{tag}_spans.jsonl", "w", encoding="utf-8") as sf, \
         open(f"{OUT}/{tag}_drawings.jsonl", "w", encoding="utf-8") as df:
        for pno in range(len(doc)):
            page = doc[pno]
            d = page.get_text("dict")
            for blk in d.get("blocks", []):
                if blk.get("type") != 0: continue
                for line in blk.get("lines", []):
                    dv = line.get("dir", (1, 0))          # לזיהוי טקסט מסובב
                    for sp in line.get("spans", []):
                        if not sp.get("text", "").strip(): continue
                        rec = {"p": pno + 1,
                               "b": [round(v, 2) for v in sp["bbox"]],   # [x0,y0,x1,y1] ב-pt
                               "s": round(sp["size"], 2), "f": sp["font"],
                               "t": sp["text"], "dir": [round(dv[0],3), round(dv[1],3)]}
                        sf.write(json.dumps(rec, ensure_ascii=False) + "\n")
            for dr in page.get_drawings():                # עיטורים/מסגרות/קווים
                r = dr.get("rect")
                if r is None: continue
                df.write(json.dumps({"p": pno+1, "r": [round(v,2) for v in r],
                    "kinds": "/".join(sorted(set(it[0] for it in dr.get("items",[])))),
                    "fill": str(dr.get("fill"))[:40], "sw": dr.get("width")}, ensure_ascii=False) + "\n")

שלב 2 — גלאי חפיפות טקסט (הליבה)

# -*- coding: utf-8 -*-
# שלב 2 — גלאי חפיפות טקסט (issue «חפיפות»): זוגות spans שחיתוך ה-bbox שלהם
# >= 6pt^2 או >= 25% מהקטן — למעט שכנים באותו פס-שורה (אותו גופן/גודל/y)
import json
from collections import defaultdict

def inter(a, b):
    x0=max(a[0],b[0]); x1=min(a[2],b[2]); y0=max(a[1],b[1]); y1=min(a[3],b[3])
    return max(0,x1-x0)*max(0,y1-y0)

def area(a): return max(0,(a[2]-a[0]))*max(0,(a[3]-a[1]))

def overlaps(spans):
    by_page = defaultdict(list)
    for s in spans: by_page[s["p"]].append(s)
    pairs = defaultdict(int); pages = defaultdict(int); total = 0
    for p, sps in by_page.items():
        n = len(sps)
        for i in range(n):
            for j in range(i+1, n):
                a, b = sps[i], sps[j]
                same_line = (abs(a["b"][1]-b["b"][1]) < 2.0 and a["f"] == b["f"]
                             and abs(a["s"]-b["s"]) < 0.3)
                if same_line: continue
                ov = inter(a["b"], b["b"])
                if ov <= 0: continue
                small = min(area(a["b"]), area(b["b"]))
                if ov >= 6.0 or (small > 0 and ov/small >= 0.25):
                    total += 1; pages[p] += 1
                    pairs[f'{a["f"]}@{a["s"]:.0f}+{b["f"]}@{b["s"]:.0f}'] += 1
    return total, pairs, pages

for tag in ["old", "new"]:
    spans = [json.loads(l) for l in open(f"{tag}_spans.jsonl")]
    t, cats, pgs = overlaps(spans)
    print(tag, "total:", t, "| pages:", len(pgs))
    for k, v in sorted(cats.items(), key=lambda kv: -kv[1])[:10]:
        print("   ", k, v)

שלב 3 — זהות גופנים פיזית (חשוב: סדר על name-table, לא על בייטים)

# -*- coding: utf-8 -*-
# שלב 3 — זהות גופנים פיזית: חילוץ ה-FontFile2 המוטמע והשוואת SHA-256 + name-table.
# חשוב: הקבצים הם SUBSET — בייטים שונים אינם בהכרח פנים שונים; סדו רק לפי שם המשפחה!
import fitz, hashlib
from fontTools.ttLib import TTFont

def extract(path, outdir, tag):
    doc = fitz.open(path); seen = {}
    for pno in range(doc.page_count):
        for f in doc[pno].get_fonts(full=True):
            xref, ext, ftype, basefont = f[0], f[1], f[2], f[3]
            if basefont in seen: continue
            name, extn, ftype2, buf = doc.extract_font(xref)
            if not buf: continue
            with open(f"{outdir}/{tag}_{basefont}.{extn}", "wb") as fh:
                fh.write(buf)
            seen[basefont] = hashlib.sha256(buf).hexdigest()[:12]
    return seen

old = extract("old_engine.pdf", "fonts_old", "old")
new = extract("new_engine.pdf", "fonts_new", "new")
for bf in sorted(set(old) | set(new)):
    print(bf, old.get(bf, "—"), new.get(bf, "—"))
# ואז לכל קובץ: קריאת זהות ה-typeface מטבלת השם
f = TTFont(f"fonts_old/old_ToraEx5.ttf", lazy=True)

# -*- coding: utf-8 -*-
# שלב 4 — מילות חלון: מלאי, מיקום, עטיפה ומרווחים
# (א) מלאי: מולטיסט מילים מכל ה-ToraWin@>=17 — חייב להיות זהה לחלוטין
# (ב) יישור: dy בין ראש מילת-החלון לראש שורת הגוף שלצידה (באותו פס) — פער זהה
# (ג) מרווח פסקה: פער בין תחתית השורה שמעל לבין ראש המילה
from collections import Counter
wins_old = [s for s in old_spans if s["f"] == "ToraWin" and s["s"] >= 17]
wins_new = [s for s in new_spans if s["f"] == "ToraWin" and s["s"] >= 17]
assert Counter(s["t"] for s in wins_old) == Counter(s["t"] for s in wins_new)

def gap_above(spans, wins):
    by_page = defaultdict(list)
    for s in spans: by_page[s["p"]].append(s)
    gaps = []
    for w in wins:
        best = None
        for s in by_page[w["p"]]:
            if s["f"] in ("ToraBook", "ToraFn"):
                sb = s["b"]
                if sb[3] <= w["b"][1]+5 and sb[3] >= w["b"][1]-40 \
                   and sb[0] < w["b"][2] and sb[2] > w["b"][0]:
                    if best is None or sb[3] > best: best = sb[3]
        if best is not None: gaps.append(w["b"][1] - best)
    gaps.sort(); return gaps

כללי פרשנות שנלמדו בסבבים

  • שמות הגופנים ב-PDF הם כינויי-סלוט (ToraEx1..9 לפי אינדקס בכותב) — יש לזהות פנים רק דרך name-table של הגופן המוטמע; בייטים שונים פירושם subsets שונים.
  • פיצול-span אינו שינוי תוכן: החדש פולט מילה-מילה, הישן פלט משפטים — לכן מספרי תווים לפי span מטעים; רק מולטיסט תווים גלובלי אמין.
  • הסחות עימוד מחייבות התאמה לפי תוכן (מילים נדירות / עוגני פרק) ולא לפי מספר עמוד, לפני כל השוואת מיקום.
  • יצוא ה-HTML הוא אותו PageModel — חפיפה שנמצאת גם ב־data-model-y מוכיחה שהשורש בבונה המודל ולא בכותב ה-PDF.
  • מחרוזות «מנוע» (כמו «(המשך)») — יש לבדוק את מקורן בטקסט המקור לפני הכרזה על תוכן חסר; הן עשויות להיות פיצ'ר יצירה של אחד המנועים.
